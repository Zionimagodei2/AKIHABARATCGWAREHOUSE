   259→    id: row.id as string,

   364→function safeParseJSON(val: unknown): string[] {
   365→  if (Array.isArray(val)) return val;
   366→  if (typeof val === "string") {
   367→    try {
   368→      const parsed = JSON.parse(val);
   369→      return Array.isArray(parsed) ? parsed : [];
   370→    } catch {
   371→      return [];
   372→    }
   373→  }
   374→  return [];
   375→}

   384→function formatDiscount(price: number, originalPrice: number): number {
   385→  if (!originalPrice || originalPrice <= 0) return 0;
   386→  return Math.round(((originalPrice - price) / originalPrice) * 100);

   389→function renderStars(rating: number, size = 14) {
   390→  const full = Math.floor(rating);
   391→  const hasHalf = rating % 1 >= 0.3;
   392→  const stars: React.ReactNode[] = [];
   393→  for (let i = 0; i < 5; i++) {
   394→    if (i < full) {
   395→      stars.push(<Star key={i} size={size} className="fill-yellow-400 text-yellow-400" />);
   396→    } else if (i === full && hasHalf) {
   397→      stars.push(<StarHalf key={i} size={size} className="fill-yellow-400 text-yellow-400" />);
   398→    } else {
   399→      stars.push(<Star key={i} size={size} className="text-gray-300" />);
   400→    }
   401→  }
   402→  return <div className="flex items-center gap-0.5">{stars}</div>;
   403→}

   405→function formatDate(dateStr: string): string {
   406→  try {
   407→    return new Date(dateStr).toLocaleDateString("en-US", {
   408→      year: "numeric",
   409→      month: "short",
   410→      day: "numeric",

   412→  } catch {

   424→      minute: "2-digit",

   431→function truncate(str: string, len: number): string {
   432→  if (!str) return "";
   433→  return str.length > len ? str.substring(0, len) + "..." : str;
   434→}

   436→function getInitials(name: string | null): string {
   437→  if (!name) return "??";
   438→  return name
   439→    .split(" ")
   440→    .map((n) => n[0])
   441→    .join("")
   442→    .toUpperCase()
   443→    .slice(0, 2);
   444→}
